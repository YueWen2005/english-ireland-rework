# ============================================================
# ire.1 边境公投 - 北爱的选择
# ============================================================
country_event = {
	id = ire.1
	title = ire.1.t
	desc = ire.1.d
	picture = GFX_report_event_generic_sign_treaty1
	is_triggered_only = yes

	immediate = {
		ENG = { add_opinion_modifier = { target = ROOT modifier = IRE_opinion_neg40 } }
		if = {
			limit = { NOT = { has_country_flag = IRE_border_done } }
			set_country_flag = IRE_border_done
		}
	}

	option = {
		name = ire.1.a
		IRE_core_northern_ireland = yes
		country_event = { id = ire.2 days = 2 }
	}

	option = {
		name = ire.1.b
		IRE_core_northern_ireland = yes
		add_stability = 0.03
		country_event = { id = ire.2 days = 4 }
	}
}

# ire.10 纽芬兰公投
# ============================================================
country_event = {
	id = ire.10
	title = ire.10.t
	desc = ire.10.d
	picture = GFX_report_event_generic_election
	is_triggered_only = yes

	immediate = {
		if = {
			limit = { ENG = { has_war = yes } }
			set_variable = { IRE_nf_chance = 1 }
		}
		else = {
			set_variable = { IRE_nf_chance = 0 }
		}
	}

	option = {
		name = ire.10.a
		trigger = {
			check_variable = { IRE_nf_chance = 1 }
		}
		custom_effect_tooltip = ire.10.a_tt
		add_political_power = 50
	}

	option = {
		name = ire.10.b
		trigger = {
			check_variable = { IRE_nf_chance = 0 }
		}
		random_list = {
			60 = {
				ENG = { declare_war_on = { target = ROOT type = annex_everything } }
			}
			40 = { add_political_power = 50 }
		}
	}
}

# ============================================================
# ire.60 1940 年 6 月：伦敦的统一提议
# 历史原型：1940 年 6 月英国曾提出「统一换参战」，
# 德瓦莱拉以「不能拿六郡换战争」为由拒绝。
# ============================================================
country_event = {
	id = ire.60
	title = ire.60.t
	desc = ire.60.d
	picture = GFX_report_event_generic_sign_treaty1
	is_triggered_only = yes

	option = {
		name = ire.60.a
		custom_effect_tooltip = ire.60.a_tt
		if = {
			limit = { 119 = { is_owned_by = ENG } }
			ENG = { transfer_state = 119 }
		}
		if = {
			limit = { owns_state = 119 }
			119 = { add_core_of = ROOT }
		}
		if = {
			limit = { exists = ENG }
			add_to_faction = ENG
		}
		add_ideas = IRE_unity_for_ports
		add_war_support = 0.10
		add_stability = -0.05
	}

	option = {
		name = ire.60.b
		add_political_power = 50
		add_stability = 0.03
	}
}
