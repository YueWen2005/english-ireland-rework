# ============================================================
# ire.1 边境公投 - 结果揭晓（支持度分档）
# ≥90 英国被迫承认，和平移交六郡
# 70-89 伦敦拒绝：拿到核心+宣称，对英战争需自行正当化
# <70  公投失败：一无所获
# ============================================================
country_event = {
	id = ire.1
	title = ire.1.t
	desc = ire.1.d
	picture = GFX_report_event_generic_sign_treaty1
	is_triggered_only = yes

	immediate = {
		ENG = { add_opinion_modifier = { target = ROOT modifier = IRE_opinion_neg40 } }
		if = {
			limit = { NOT = { has_country_flag = IRE_border_done } }
			set_country_flag = IRE_border_done
		}
	}

	# 公布结果，要求承认合并
	option = {
		name = ire.1.a
		trigger = {
			check_variable = { IRE_plebiscite_support > 69 }
		}
		IRE_core_northern_ireland = yes
		# 伦敦震怒：经济制裁（英国拒绝承认时）
		if = {
			limit = { check_variable = { IRE_plebiscite_support < 90 } }
			country_event = { id = ire.2 days = 2 }
		}
	}

	# 承认公投无效，维持现状
	option = {
		name = ire.1.b
		trigger = {
			check_variable = { IRE_plebiscite_support < 70 }
		}
		add_political_power = 50
		add_stability = -0.03
		add_ideas = IRE_north_unintegrated
	}
}

# ============================================================
# ire.10 纽芬兰公投（已删除"60% 概率英国随机宣战"——
# 小国主动挑衅不应导致随机亡国；成败取决于支持度与英国是否忙于大战）
# ============================================================
country_event = {
	id = ire.10
	title = ire.10.t
	desc = ire.10.d
	picture = GFX_report_event_generic_election
	is_triggered_only = yes

	# 稳妥路线：英国深陷大战自顾不暇，或支持度已臻成熟——公投必然通过
	option = {
		name = ire.10.a
		trigger = {
			OR = {
				ENG = { has_war = yes }
				check_variable = { IRE_nf_support > 79 }
			}
		}
		custom_effect_tooltip = ire.10.a_tt
		set_country_flag = IRE_nf_yes
		add_political_power = 25
	}

	# 强行公投：条件不足时赌一把——50% 成功，失败则伦敦记恨
	option = {
		name = ire.10.b
		trigger = {
			ENG = { has_war = no }
			check_variable = { IRE_nf_support < 80 }
		}
		random_list = {
			50 = {
				custom_effect_tooltip = ire.10_success_tt
				set_country_flag = IRE_nf_yes
			}
			50 = {
				custom_effect_tooltip = ire.10_fail_tt
				ENG = { add_opinion_modifier = { target = ROOT modifier = IRE_opinion_neg40 } }
				add_stability = -0.03
				set_country_flag = IRE_nf_failed
			}
		}
	}
}
