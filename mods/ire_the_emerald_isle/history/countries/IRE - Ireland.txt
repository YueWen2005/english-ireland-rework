#   汤姆·巴里     → 《志愿军团》

oob = "IRE_units"

# 开局科技
# 原版给的四项（infantry_weapons / tech_trucks / fuel_silos / basic_train）必须保留：
#   tech_trucks  —— 没有它开局不能造卡车，摩托化完全无从谈起
#   fuel_silos   —— 没有它无燃油储备，飞机与机械化部队直接趴窝
#   basic_train  —— 《铁路车辆生产计划》决议的 has_tech 门槛就是它
set_technology = {
	infantry_weapons = 1
	infantry_weapons1 = 1
	tech_trucks = 1
	fuel_silos = 1
	basic_train = 1

	# MOD 追加：支援武器与工业工具，避免开局造不出师
	support_weapons = 1
	tech_support = 1
	basic_machine_tools = 1
	improved_machine_tools = 1

	# MOD 追加：火炮。
	# 《圣帕特里克营》《野鹅军团》两个师模板都带 1 个炮兵营，
	# 但 gw_artillery 才是解锁 artillery_equipment_1 与 field_guns 兵种的科技。
	# 没有它，这些「精锐师」的炮兵营永远补不满，爱尔兰也造不出任何火炮。
	# 历史上爱尔兰陆军自内战起就保有 18 磅野战炮，1936 年给这一项是合理的。
	gw_artillery = 1

	early_fighter = 1
}

# 开局科研槽：原版就是 3 个，这里显式写出以免日后漏改
set_research_slots = 3

# 开局运输船
# 原版 10 艘。MOD 的 A 线有大量跨大西洋内容（商船队精神、南锥体补给线、
# 新世界远征、跨大西洋空桥），10 艘不够周转，给到 20 艘
set_convoys = 20

# 开局装备库存
# 两个开局师是 50% 装备率；火炮库存是为了让《圣帕特里克营》《野鹅军团》
# 这类带炮兵营的精锐师模板在 create_unit 时能真正拿到炮，而不是空着编制。
add_equipment_to_stockpile = { type = infantry_equipment_1 amount = 400 }
add_equipment_to_stockpile = { type = support_equipment_1 amount = 200 }
add_equipment_to_stockpile = { type = artillery_equipment_1 amount = 80 }

# 开局政治：爱尔兰是议会民主国，中立是外交政策而非执政意识形态
# last_election / election_frequency 沿用原版，否则选举周期会退回默认值

	election_frequency = 48

}
