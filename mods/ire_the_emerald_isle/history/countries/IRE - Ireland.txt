# 开局科技
# 原版给的四项（infantry_weapons / tech_trucks / fuel_silos / basic_train）必须保留：
#   tech_trucks  —— 没有它开局不能造卡车，摩托化完全无从谈起
#   fuel_silos   —— 没有它无燃油储备，飞机与机械化部队直接趴窝
#   basic_train  —— 《铁路车辆生产计划》决议的 has_tech 门槛就是它

	infantry_weapons = 1
	infantry_weapons1 = 1
	tech_trucks = 1
	fuel_silos = 1
	basic_train = 1

	# MOD 追加：支援武器与工业工具，避免开局造不出师

	tech_support = 1
	basic_machine_tools = 1
	improved_machine_tools = 1

	# 《圣帕特里克营》《野鹅军团》两个师模板都带 1 个炮兵营，
	# 但 gw_artillery 才是解锁 artillery_equipment_1 与 field_guns 兵种的科技。
	# 没有它，这些「精锐师」的炮兵营永远补不满，爱尔兰也造不出任何火炮。
	# 历史上爱尔兰陆军自内战起就保有 18 磅野战炮，1936 年给这一项是合理的。

	early_fighter = 1
