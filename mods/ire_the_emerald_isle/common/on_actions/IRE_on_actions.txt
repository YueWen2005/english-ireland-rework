on_actions = {
	on_startup = {
		effect = {
			# on_startup 无国家作用域，需用 every_country 包裹
			every_country = {
				limit = { tag = IRE }
				IRE_init_variables = yes
				# 民主议会制开局：确保德·瓦莱拉是执政领袖
				IRE_Éamon_de_valera = { promote_leader = yes }
			}
		}
	}

	on_daily = {
		effect = {
			if = {
				limit = { tag = IRE }
				# 紧急状态 1→2→3 级自动升级/降级
				IRE_emergency_tier_check = yes

				# 保卫爱尔兰：按投降进度维护 1-5 级精神
				IRE_defend_ireland_check = yes

				# ---- C红线：英国已社会主义时跳过内战并合并 ----
				if = {
					limit = {
						has_country_flag = IRE_red_path
						has_completed_focus = IRE_R_end_eve
						country_exists = ENG
						ENG = { has_government = communism }
						NOT = { has_country_flag = IRE_red_britain_merged }
					}
					if = {
						limit = { NOT = { has_completed_focus = IRE_R_end_ignite } }
						complete_national_focus = IRE_R_end_ignite
					}
					if = {
						limit = { ENG = { has_civil_war = no } }
						if = {
							limit = { NOT = { has_completed_focus = IRE_R_end_union } }
							complete_national_focus = IRE_R_end_union
						}
						IRE_R_merge_red_britain = yes
					}
				}

				# ---- C红线：吹红途中英国提前对爱宣战 → 英国爆发低强度内战 ----
				if = {
					limit = {
						has_country_flag = IRE_red_path
						has_completed_focus = IRE_R_end_red_britain
						NOT = { has_completed_focus = IRE_R_end_ulster }
						NOT = { has_country_flag = IRE_eng_early_revolution }
						country_exists = ENG
						ENG = {
							exists = yes
							NOT = { has_government = communism }
							has_civil_war = no
							has_war_with = IRE
						}
					}
					set_country_flag = IRE_eng_early_revolution
					ENG = {
						start_civil_war = { ideology = communism size = 0.25 }
					}
				}

				# ---- 公投支持度：稳定度 ≥ 60% 一次性 +10 ----
				if = {
					limit = {
						NOT = { has_country_flag = IRE_border_done }
						has_stability > 0.6
						NOT = { has_country_flag = IRE_stability_bonus_given }
					}
					add_to_variable = { IRE_plebiscite_support = 10 }
					set_country_flag = IRE_stability_bonus_given
				}

				# ---- 圣帕特里克日：每 365 天一次（flag 过期后自动再触发）----
				if = {
					limit = {
						date > 1936.3.16
						NOT = { has_country_flag = IRE_st_patricks_year }
					}
					set_country_flag = { flag = IRE_st_patricks_year days = 365 value = 1 }
					country_event = { id = ire.50 days = 0 }
				}
			}
		}
	}

	on_monthly = {
		effect = {
			if = {
				limit = { tag = IRE has_idea = IRE_armed_neutrality_idea }
				army_experience = 2
			}

			# ---- 1940 年 5 月：戈茨空降 ----
			if = {
				limit = {
					tag = IRE
					date > 1940.4.30
					date < 1940.9.1
					NOT = { has_country_flag = IRE_goertz_done }
					NOT = { has_war_with = GER }
				}
				set_country_flag = IRE_goertz_done
				country_event = { id = ire.63 days = 0 }
			}

			# ---- 1941 年 4 月：贝尔法斯特大空袭 ----
			if = {
				limit = {
					tag = IRE
					date > 1941.3.31
					date < 1941.9.1
					NOT = { has_country_flag = IRE_belfast_blitz_done }
					119 = { is_owned_by = ENG }
					ENG = { has_war = yes }
				}
				set_country_flag = IRE_belfast_blitz_done
				country_event = { id = ire.61 days = 0 }
			}

			# ---- 1941 年 5 月 31 日：北斯特兰德轰炸 ----
			if = {
				limit = {
					tag = IRE
					date > 1941.5.31
					date < 1941.9.1
					NOT = { has_country_flag = IRE_north_strand_done }
					NOT = { has_war_with = GER }
				}
				set_country_flag = IRE_north_strand_done
				country_event = { id = ire.62 days = 0 }
			}

			# ---- 1944 年 4 月：美国照会 ----
			if = {
				limit = {
					tag = IRE
					date > 1944.3.31
					date < 1944.9.1
					NOT = { has_country_flag = IRE_american_note_done }
					NOT = { has_war_with = USA }
				}
				set_country_flag = IRE_american_note_done
				country_event = { id = ire.64 days = 0 }
			}

			# ---- 1945 年 5 月：希特勒之死 ----
			if = {
				limit = {
					tag = IRE
					date > 1945.4.30
					date < 1945.9.1
					NOT = { has_country_flag = IRE_hitler_death_done }
				}
				set_country_flag = IRE_hitler_death_done
				country_event = { id = ire.65 days = 0 }
			}

			# ---- 黑线：曝光度月增与阈值（IRA 内战线）----
			if = {
				limit = {
					tag = IRE
					has_country_flag = IRE_dark_path
					NOT = { has_country_flag = IRE_dark_surrender }
				}
				add_to_variable = { IRE_exposure = 5 }
				if = {
					limit = { check_variable = { IRE_exposure > 100 } }
					set_variable = { IRE_exposure = 100 }
				}
				if = {
					limit = {
						check_variable = { IRE_exposure > 49 }
						NOT = { has_country_flag = IRE_exposure_warned }
					}
					set_country_flag = IRE_exposure_warned
					country_event = { id = ire.93 days = 0 }
				}
				if = {
					limit = {
						check_variable = { IRE_exposure > 89 }
						NOT = { has_country_flag = IRE_exposure_crisis }
					}
					set_country_flag = IRE_exposure_crisis
					country_event = { id = ire.95 days = 0 }
				}
			}

			# ---- 黑线：1940.8.14 拉塞尔之死（史实锚点，不可跳过）----
			if = {
				limit = {
					tag = IRE
					has_country_flag = IRE_dark_path
					date > 1940.8.13
					date < 1940.12.1
					NOT = { has_country_flag = IRE_russell_dead }
				}
				set_country_flag = IRE_russell_dead
				country_event = { id = ire.94 days = 0 }
			}

			# ---- 红线：团结值 40-70 区间的宗派危机（随机）----
			if = {
				limit = {
					tag = IRE
					has_country_flag = IRE_red_path
					check_variable = { IRE_red_unity > 39 }
					check_variable = { IRE_red_unity < 71 }
					NOT = { has_country_flag = IRE_unity_crisis_done }
				}
				random = {
					chance = 15
					set_country_flag = IRE_unity_crisis_done
					country_event = { id = ire.103 days = 0 }
				}
			}

			# ---- 红线：1937.7 十字架与红旗 ----
			if = {
				limit = {
					tag = IRE
					has_country_flag = IRE_red_path
					date > 1937.6.30
					date < 1937.12.1
					NOT = { has_country_flag = IRE_constitution_clash_done }
				}
				set_country_flag = IRE_constitution_clash_done
				country_event = { id = ire.102 days = 0 }
			}

			# ---- 红线：1944.2 德累斯顿的幽灵 ----
			if = {
				limit = {
					tag = IRE
					has_country_flag = IRE_red_path
					date > 1944.1.31
					date < 1944.6.1
					NOT = { has_country_flag = IRE_dresden_done }
				}
				set_country_flag = IRE_dresden_done
				country_event = { id = ire.107 days = 0 }
			}

			# ---- 红线：西班牙黄金（1936.10 起比斯开湾航线，红线开局后触发）----
			if = {
				limit = {
					tag = IRE
					has_country_flag = IRE_red_path
					has_completed_focus = IRE_R_connolly_column
					SPA = { has_volunteers_amount_from = { tag = ROOT count > 0 } }
					date > 1936.10.14
					date < 1939.1.1
					NOT = { has_country_flag = IRE_gold_done }
				}
				set_country_flag = IRE_gold_done
				country_event = { id = ire.108 days = 0 }
			}

			# ---- A线外交环境：美国租借与爱尔兰游说 ----
			if = {
				limit = {
					tag = IRE
					has_country_flag = IRE_campaign_started
					country_exists = USA
					USA = { has_war = yes }
					has_variable = IRE_diaspora_network
					check_variable = { IRE_diaspora_network > 39 }
					NOT = { has_country_flag = IRE_env_us_lobby_done }
				}
				set_country_flag = IRE_env_us_lobby_done
				country_event = { id = ire_nw.80 days = 0 }
			}

			# ---- A线外交环境：加拿大归并后的英联邦裂痕 ----
			if = {
				limit = {
					tag = IRE
					has_country_flag = IRE_campaign_started
					has_country_flag = IRE_canada_annexed
					country_exists = ENG
					any_country = {
						OR = { tag = AST tag = NZL tag = SAF tag = RAJ }
						exists = yes
					}
					NOT = { has_country_flag = IRE_env_commonwealth_done }
				}
				set_country_flag = IRE_env_commonwealth_done
				country_event = { id = ire_nw.81 days = 0 }
			}

			# ---- A线外交环境：法国沦陷与自由法国请求 ----
			if = {
				limit = {
					tag = IRE
					has_country_flag = IRE_campaign_started
					country_exists = FRA
					FRA = { has_capitulated = yes }
					NOT = { has_country_flag = IRE_env_free_france_done }
				}
				set_country_flag = IRE_env_free_france_done
				country_event = { id = ire_nw.82 days = 0 }
			}

			# ---- A线外交环境：东线志愿医疗队 ----
			if = {
				limit = {
					tag = IRE
					has_country_flag = IRE_campaign_started
					country_exists = SOV
					country_exists = GER
					SOV = { has_war_with = GER }
					has_variable = IRE_diaspora_network
					check_variable = { IRE_diaspora_network > 39 }
					NOT = { has_country_flag = IRE_env_eastern_front_done }
				}
				set_country_flag = IRE_env_eastern_front_done
				country_event = { id = ire_nw.83 days = 0 }
			}

			# ---- A线外交环境：美国参战后的安全保障 ----
			if = {
				limit = {
					tag = IRE
					has_country_flag = IRE_campaign_started
					country_exists = USA
					USA = { has_war = yes }
					OR = {
						has_completed_focus = IRE_A_tammany_hall
						has_completed_focus = IRE_A_green_dollar
					}
					NOT = { has_country_flag = IRE_env_us_guarantee_done }
				}
				set_country_flag = IRE_env_us_guarantee_done
				country_event = { id = ire_nw.84 days = 0 }
			}

			# ---- A线外交环境：战后欧洲秩序 ----
			if = {
				limit = {
					tag = IRE
					has_country_flag = IRE_campaign_started
					has_completed_focus = IRE_A_green_international
					date > 1944.1.1
					NOT = { any_country = { has_war = yes is_major = yes } }
					NOT = { has_country_flag = IRE_env_postwar_done }
				}
				set_country_flag = IRE_env_postwar_done
				country_event = { id = ire_nw.85 days = 0 }
			}
		}
	}
}
