### 初始化变量
IRE_init_variables = {
	if = {
		limit = { NOT = { has_variable = IRE_plebiscite_support } }
		set_variable = { IRE_plebiscite_support = 45 }
	}
	if = {
		limit = { NOT = { has_variable = IRE_unity_campaign_count } }
		set_variable = { IRE_unity_campaign_count = 0 }
	}
	if = {
		limit = { NOT = { has_variable = IRE_neutral_credibility } }
		set_variable = { IRE_neutral_credibility = 50 }
	}
}

### 决议用的小效果
IRE_add_plebiscite_support_12 = {
	add_to_variable = { IRE_plebiscite_support = 12 }
}
