# 北爱尔兰结算（state 119 = 北爱尔兰）
# IRE 开局就有 119 州核心：公投通过（>69）即直接移交，伦敦只保留抗议/制裁
IRE_core_northern_ireland = {
	if = {
		limit = { NOT = { has_variable = IRE_plebiscite_support } }
		set_variable = { IRE_plebiscite_support = 45 }
	}

	# 无论结果如何，先获得北爱领土宣称
	119 = { add_claim_by = ROOT }

	# 公投通过：直接移交六郡；支持度越高，伦敦越不敢阻拦
	if = {
		limit = { check_variable = { IRE_plebiscite_support > 69 } }
		if = {
			limit = { 119 = { NOT = { is_owned_by = ROOT } } }
			ROOT = { transfer_state = 119 }
		}
		119 = { add_core_of = ROOT }
		if = {
			limit = { check_variable = { IRE_plebiscite_support > 89 } }
			add_stability = 0.10
			add_political_power = 50
		}
		else = {
			add_stability = 0.05
		}
	}
	# 支持度 <70：公投失败——什么也拿不到，仅保留宣称
	else = {
		add_stability = -0.05
		add_political_power = -50
	}
}

IRE_emergency_tier_check = {
	# 一级 → 三级：直接开战（跳级）
	if = {
		limit = {
			has_idea = IRE_emergency_powers
			has_war = yes
		}
		remove_ideas = IRE_emergency_powers
		add_ideas = IRE_emergency_powers_3
	}
	# 一级 → 二级：大战爆发但自己尚未参战
	else_if = {
		limit = {
			has_idea = IRE_emergency_powers
			has_war = no
			any_other_country = { is_major = yes has_war = yes }
			date > 1939.6.1
		}
		remove_ideas = IRE_emergency_powers
		add_ideas = IRE_emergency_powers_2
	}
	# 二级 → 三级：参战
	else_if = {
		limit = {
			has_idea = IRE_emergency_powers_2
			has_war = yes
		}
		remove_ideas = IRE_emergency_powers_2
		add_ideas = IRE_emergency_powers_3
	}
	# 三级 → 二级：战争结束后降级
	else_if = {
		limit = {
			has_idea = IRE_emergency_powers_3
			has_war = no
			date > 1945.1.1
		}
		remove_ideas = IRE_emergency_powers_3
		add_ideas = IRE_emergency_powers_2
	}
}

			limit = { has_war = yes }
			if = {
				limit = { is_ai = no NOT = { has_country_flag = IRE_foggy_dew_played } surrender_progress > 0.29 }
				set_country_flag = IRE_foggy_dew_played
				play_song = "ire_a_foggy_dew"

# E 吞并大英前的继承处理（ire.114 黑线分赃 / IRE_R_merge_red_britain 红线合并共用）
#   1) 非洲殖民地（非核心）独立为「英属非洲」BWA
#   2) 大英将领全部并入爱尔兰军队
#   3) 大英核心领土同步成为爱尔兰核心
#   必须在 annex_country 之前调用（否则 ENG 已消亡、将领无法转移）
# ============================================================
IRE_annex_britain_inherit = {
	every_country = {
		limit = {
			original_tag = ENG
			NOT = { tag = IRE }
			exists = yes
		}
		# 非洲殖民地独立
		every_owned_state = {
			limit = {
				is_on_continent = africa
				NOT = { is_core_of = PREV }
			}
			set_state_owner_to = BWA
		}
		# 将领继承
		every_unit_leader = {
			set_nationality = IRE
		}
		# 核心继承
		every_owned_state = {
			limit = { is_core_of = PREV }
			add_core_of = IRE
		}
	}
}
