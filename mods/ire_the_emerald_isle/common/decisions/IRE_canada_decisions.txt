IRE_canada_cat = {

	# 国土开发：每期覆盖全部可开发省份
	# ------------------------------------------------------------
	IRE_canada_development_plan = {
		icon = generic_construction
		cost = 75
		days_remove = 150
		fire_only_once = no

		visible = {
			has_country_flag = IRE_canada_annexed
			any_owned_state = { IRE_canadian_state = yes }
		}

		available = {
			has_country_flag = IRE_canada_annexed
			any_owned_state = { IRE_canadian_developable_state = yes }
			OR = {
				NOT = { has_variable = IRE_canada_dev_count }
				check_variable = { IRE_canada_dev_count < 3 }
			}
		}

		complete_effect = {
			if = {
				limit = { NOT = { has_variable = IRE_canada_dev_count } }
				set_variable = { IRE_canada_dev_count = 0 }
			}
			add_to_variable = { IRE_canada_dev_count = 1 }
			hidden_effect = {
				every_owned_state = {
					limit = { IRE_canadian_developable_state = yes }
					add_extra_state_shared_building_slots = 1
					add_building_construction = { type = infrastructure level = 1 instant_build = yes }
				}
				if = {
					limit = { any_owned_state = { IRE_canadian_developable_state = yes } }
					random_owned_controlled_state = {
						limit = { IRE_canadian_developable_state = yes }
						add_building_construction = { type = industrial_complex level = 1 instant_build = yes }
					}
				}
				add_offsite_building = { type = industrial_complex level = 1 }
				add_manpower = 10000
				add_stability = 0.02
				if = {
					limit = { check_variable = { IRE_canada_dev_count = 3 } }
					add_offsite_building = { type = arms_factory level = 1 }
					add_manpower = 15000
				}
			}
			custom_effect_tooltip = IRE_tt_canada_development
			custom_effect_tooltip = IRE_tt_canada_development_offsite
			if = {
				limit = { check_variable = { IRE_canada_dev_count = 3 } }
				custom_effect_tooltip = IRE_tt_canada_development_final
			}
		}

		ai_will_do = { base = 60 }
	}
