	IRE_unity_campaign_cat = {

		IRE_unity_campaign = {
			icon = generic_political_rally
			cost = 50
			days_remove = 30

			available = {
				has_completed_focus = IRE_border_plebiscite
				custom_trigger_tooltip = {
					tooltip = IRE_tt_border_not_done
									NOT = { has_country_flag = IRE_border_done }
				}
				check_variable = { IRE_unity_campaign_count < 5 }
			}

			visible = {
				has_completed_focus = IRE_border_plebiscite
				NOT = { has_country_flag = IRE_border_done }
				check_variable = { IRE_unity_campaign_count < 5 }
			}

			fire_only_once = no

		complete_effect = {
			add_to_variable = { IRE_plebiscite_support = 12 }
			if = {
				limit = { check_variable = { IRE_plebiscite_support > 100 } }
				set_variable = { IRE_plebiscite_support = 100 }
			}
			if = {
				limit = { NOT = { has_variable = IRE_unity_campaign_count } }
				set_variable = { IRE_unity_campaign_count = 0 }
			}
			add_to_variable = { IRE_unity_campaign_count = 1 }
			custom_effect_tooltip = IRE_tt_unity_campaign
		}

			ai_will_do = { base = 40 }
		}
	}
