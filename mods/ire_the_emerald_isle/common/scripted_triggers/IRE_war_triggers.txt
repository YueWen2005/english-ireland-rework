# 欧洲大国之间的战争：两个欧洲主要国家互相开战。
# 不包含意大利-埃塞俄比亚这类殖民战争，也不要求爱尔兰参战。
IRE_european_major_war = {
	any_country = {
		is_major = yes
		has_war = yes
		is_on_same_continent_as = ROOT
		any_country = {
			is_major = yes
			has_war = yes
			is_on_same_continent_as = ROOT
			has_war_with = PREV
		}
	}
}

# 德国陷入多线大战：与英/苏/美中的至少两个主要国家交战。
IRE_germany_widespread_war = {
	country_exists = GER
	GER = { has_war = yes }
	date > 1940.1.1
	OR = {
		AND = {
			any_country = { original_tag = ENG has_war_with = GER }
			GER = { has_war_with = SOV }
		}
		AND = {
			any_country = { original_tag = ENG has_war_with = GER }
			GER = { has_war_with = USA }
		}
		AND = {
			GER = { has_war_with = SOV }
			GER = { has_war_with = USA }
		}
	}
}

# 德国已投降/消亡：欧洲希望进入战后阶段。
IRE_germany_defeated = {
	OR = {
		NOT = { country_exists = GER }
		GER = { has_capitulated = yes }
	}
}
