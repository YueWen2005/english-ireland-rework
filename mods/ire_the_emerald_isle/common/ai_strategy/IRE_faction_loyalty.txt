FRA_stay_in_european_hope = {
	allowed = {
		original_tag = FRA
	}
	enable = {
		IRE = {
			exists = yes
			has_idea = IRE_european_hope
		}
		NOT = { has_war_with = IRE }
	}
	abort_when_not_enabled = yes

	ai_strategy = {
		type = alliance
		id = IRE
		value = 1000
	}
	ai_strategy = {
		type = pp_spend_priority
		id = become_faction_leader
		value = -1000
	}
	# 红法不要被共产国际拉走；战后也继续偏好欧洲希望
	ai_strategy = {
		type = diplo_action_acceptance
		id = IRE
		target = join_allies
		value = 1000
	}
	ai_strategy = {
		type = diplo_action_acceptance
		id = SOV
		target = join_allies
		value = -99999
	}
}
